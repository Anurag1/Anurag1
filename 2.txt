🧠 3. Orchestrator

orchestrator/app/main.py

from fastapi import FastAPI
from planner import plan
from executor import execute
app = FastAPI()
@app.post("/execute")
def run(task: dict):
    steps = plan(task)
    result = execute(steps, task)
    return {"steps": steps, "result": result}

orchestrator/app/planner.py

def plan(task):
    t = task.get("type", "")
    if "code" in t:
        return ["code"]
    elif "search" in t:
        return ["retrieval"]
    return ["reason"]

orchestrator/app/executor.py

import requests
MAP = {
    "code": "http://compute:9001/code",
    "retrieval": "http://compute:9002/search",
    "reason": "http://compute:9003/reason"
}
def execute(steps, task):
    out = []
    for s in steps:
        res = requests.post(MAP[s], json=task)
        out.append(res.json())
    return out
